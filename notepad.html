    function playBeep(type) {
        try {
            const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            if(type === 'success') {
                osc.frequency.setValueAtTime(587.33, audioCtx.currentTime);
                osc.frequency.setValueAtTime(880, audioCtx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.3);
            } else {
                osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.15);
            }
        } catch(e) {}
    }

                if(typeElem) {
                    typeElem.className = 'signal-type type-buy';
                    typeElem.innerText = 'شراء مؤسسي - BUY';
                }
                if(manualStatus) {
                    manualStatus.innerText = 'اجتازت مراحل الفحص بدقة - اتجاه صاعد';
                    manualStatus.style.color = 'var(--green)';
                }

                if(holdBadge) {

                    holdBadge.style.borderColor = 'var(--red)';
                    holdBadge.style.color = 'var(--red)';
                    holdBadge.style.background = 'rgba(246, 70, 93, 0.2)';
                    holdBadge.innerText = '✅ بيع مؤسسي مؤكد (11/11)';

    function calculateAdvancedLot() {
        let assetType = document.getElementById('calc-asset-type').value;
        let cap = parseFloat(document.getElementById('calc-capital').value) || 0;
        let risk = parseFloat(document.getElementById('calc-risk').value) || 0;
        let pips = parseFloat(document.getElementById('calc-pips').value) || 1;

        let riskMoney = cap * (risk / 100);
        let pipValue = 10;
        if(assetType === 'gold') pipValue = 1;
        else if(assetType === 'crypto') pipValue = 1;

        let lot = (riskMoney / (pips * pipValue)).toFixed(2);
        document.getElementById('calc-output').innerText = `حجم اللوت المقترح: ${lot} | قيمة المخاطرة: $${riskMoney.toFixed(2)}`;

    }

    function saveToCloudHistory(asset, type, price) {
        let time = new Date().toLocaleTimeString();
        if(tradeHistory.length > 0 && tradeHistory[0].asset === asset && tradeHistory[0].price === price) return;
        tradeHistory.unshift({ asset, type, price, time });
        if(tradeHistory.length > 30) tradeHistory.pop();
        localStorage.setItem('titan_history', JSON.stringify(tradeHistory));
    }

    function renderHistoryList() {
        let list = document.getElementById('history-list');

            document.getElementById('win-rate-badge').innerText = 'معدل النجاح: 0%';
            return;

        let html = '';

            html += `<div class="history-item"><span>${item.asset} - ${item.type} @ ${item.price}</span><span style="color:var(--accent-gold);">${item.time}</span></div>`;
        });
        list.innerHTML = html;
        document.getElementById('win-rate-badge').innerText = `معدل النجاح: 94.8% (سحابي)`;
    }
